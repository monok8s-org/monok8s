#   user:iris           — tenant:acme admin BUT denied (no read)

# ── Tenant: acme — full role matrix ──────────────────────────────────────

# Two owners — last-owner protection asserts LookupSubjects returns ≥ 1.
tenant:acme#owner@user:bob
tenant:acme#owner@user:carol

# Owner who is suspended — must lose delete + transfer_ownership.
tenant:acme#owner@user:jane

# Direct admin (full manage_*).
tenant:acme#admin@user:dave

# Lower roles.

# Suspended admin — role present, suspended subtracts → no write.
tenant:acme#admin@user:henry
tenant:acme#suspended@user:henry

# Denied admin — role present, denied subtracts → no read.
tenant:acme#admin@user:iris
tenant:acme#denied@user:iris

# ── Group: engineering (direct admin on acme) ────────────────────────────
group:engineering#tenant@tenant:acme

# ── Nested group: senior_eng inside engineering ──────────────────────────

# engineering's membership flows engineering's admin role onto liam.

group:engineering#member@group:senior_eng#membership
